No Raspberry Pi, ative o I²C nas configurações do sistema antes de usar. Para vários sensores no mesmo barramento, mantenha todos desligados pelo XSHUT, ligue um por vez e troque o endereço de cada um por software.

ESP32 (Arduino, biblioteca da Pololu)

#include <Wire.h>
#include <VL53L1X.h>   // biblioteca da Pololu

VL53L1X sensor;

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);        // SDA, SCL
  Wire.setClock(400000);
  sensor.setTimeout(500);
  if (!sensor.init()) {
    Serial.println("Sensor não encontrado");
    while (1);
  }
  sensor.setDistanceMode(VL53L1X::Long);
  sensor.setMeasurementTimingBudget(50000);
  sensor.startContinuous(50);
}

void loop() {
  uint16_t mm = sensor.read();
  if (!sensor.timeoutOccurred())
    Serial.printf("%u mm\n", mm);
}

Raspberry Pi (Python, biblioteca da Adafruit)

import time
import board
import adafruit_vl53l1x

i2c = board.I2C()
sensor = adafruit_vl53l1x.VL53L1X(i2c)
sensor.start_ranging()

while True:
    if sensor.data_ready:
        print(sensor.distance, "cm")
        sensor.clear_interrupt()
    time.sleep(0.1)
